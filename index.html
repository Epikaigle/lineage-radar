import asyncio from pyodide.http import pyfetch from bs4 import BeautifulSoup import json # Fonction pour récupérer le contenu d'une URL async def fetch_url(url): try: response = await pyfetch(url) return await response.text() except Exception as e: print(f"Erreur lors de la récupération de {url}: {e}") return None # Récupérer la liste des appareils depuis la page wiki async def get_devices(): wiki_url = "https://wiki.lineageos.org/devices/" html = await fetch_url(wiki_url) if not html: return [] soup = BeautifulSoup(html, 'html.parser') device_cards = soup.find_all('div', class_='device-card') devices = [] for card in device_cards: codename = card.get('data-codename', '') name_elem = card.find('h2') oem_elem = card.find('p', class_='device-oem') if not codename or not name_elem or not oem_elem: continue name = name_elem.text.strip() oem = oem_elem.text.strip() status = 'discontinued' if 'discontinued' in card.get('class', []) else 'supported' if oem.lower() == 'xiaomi': devices.append({'codename': codename, 'name': name, 'status': status}) return devices # Récupérer la date de la dernière mise à jour pour un appareil supporté async def get_last_build_date(codename): api_url = f"https://download.lineageos.org/api/v1/{codename}/nightly" response = await fetch_url(api_url) if not response: return None try: builds = json.loads(response) if builds and isinstance(builds, list): last_build = max(builds, key=lambda x: x.get('date', '0000-00-00')) return last_build.get('date', None) except Exception as e: print(f"Erreur pour {codename}: {e}") return None #
